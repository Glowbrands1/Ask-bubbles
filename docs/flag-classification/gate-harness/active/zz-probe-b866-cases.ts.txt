  id: `zz-probe-b866-${key}`,
  title: turns.join(" → "),
  report: "Gate-Bubbles probe (scratch, never committed)",
  status: "fixed",
  async run(t) {
    vi.setSystemTime(new Date("2026-11-12T15:00:00Z"));
    t.directory([...STAFF]);
    t.as("location_manager", WEST_DODGE);
    const thread = t.thread();
    const out: unknown[] = [];
    for (const turn of turns) {
      const last = await thread.say(turn);
      const card = thread.proposal as { templateKey?: string; issue?: string | null; supportsInlineDraft?: boolean } | null;
      const issueLine = last.content.split("\n").find((line) => line.includes("**Issue:**")) ?? null;
